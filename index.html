async function fetchLiveETA() { const statusEl = document.getElementById('status'); const lastUpdateEl = document.getElementById('lastUpdate'); let allEta = []; try { for (const stop of KMB_STOPS) { // 修正 URL const url = `https://data.etabus.gov.hk/v1/transport/kmb/stop-eta/${stop.id}?_t=${Date.now()}`; const res = await fetch(url, { cache: "no-store" }); if (!res.ok) continue; const json = await res.json(); if (json.data && Array.isArray(json.data)) { json.data.forEach(item => { if (!item.eta) return; const etaTime = new Date(item.eta); // 過濾已過站資料 if (etaTime <= new Date()) return; allEta.push({ route: item.route, dest: item.dest_tc, etaTime, co: item.co || 'KMB', stopName: stop.name }); }); } } // 按到站時間排序 allEta.sort((a, b) => a.etaTime - b.etaTime); cachedBusData = allEta; renderList(cachedBusData); statusEl.innerHTML = '● 官方 API 連線正常'; const now = new Date(); lastUpdateEl.innerText = `最後更新: ${now.toLocaleTimeString('zh-HK', { hour12: false })}`; } catch (err) { console.error(err); statusEl.innerHTML = '▲ 數據更新異常'; } }
